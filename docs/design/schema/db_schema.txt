------------------------------------------------------------
USERS
------------------------------------------------------------

CREATE TABLE users (
    user_id INTEGER PRIMARY KEY,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    display_name TEXT,
    created_at TEXT NOT NULL
);

------------------------------------------------------------
HOUSEHOLDS
------------------------------------------------------------

CREATE TABLE households (
    household_id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    join_code TEXT UNIQUE NOT NULL,
    occupant_count INTEGER NOT NULL,
    dwelling_type TEXT,
    postcode TEXT,
    created_at TEXT NOT NULL
);

------------------------------------------------------------
HOUSEHOLD MEMBERSHIPS
------------------------------------------------------------

CREATE TABLE household_memberships (
    membership_id INTEGER PRIMARY KEY,
    user_id INTEGER NOT NULL,
    household_id INTEGER NOT NULL,
    role TEXT NOT NULL,
    joined_at TEXT NOT NULL,
    left_at TEXT,

    FOREIGN KEY (user_id)
        REFERENCES users(user_id),

    FOREIGN KEY (household_id)
        REFERENCES households(household_id),

    UNIQUE (user_id, household_id, joined_at)
);

------------------------------------------------------------
ENERGY ENTRIES
------------------------------------------------------------

CREATE TABLE energy_entries (
    energy_entry_id INTEGER PRIMARY KEY,
    household_id INTEGER NOT NULL,
    period_start TEXT NOT NULL,
    period_end TEXT,
    electricity_kwh REAL,
    solar_generation_kwh REAL,
    cost REAL,
    created_at TEXT NOT NULL,

    FOREIGN KEY (household_id)
        REFERENCES households(household_id)
);

------------------------------------------------------------
WATER ENTRIES
------------------------------------------------------------

CREATE TABLE water_entries (
    water_entry_id INTEGER PRIMARY KEY,
    household_id INTEGER NOT NULL,
    period_start TEXT NOT NULL,
    period_end TEXT,
    water_litres REAL,
    cost REAL,
    created_at TEXT NOT NULL,

    FOREIGN KEY (household_id)
        REFERENCES households(household_id)
);

------------------------------------------------------------
WASTE ENTRIES
------------------------------------------------------------

CREATE TABLE waste_entries (
    waste_entry_id INTEGER PRIMARY KEY,
    household_id INTEGER NOT NULL,
    period_start TEXT NOT NULL,
    period_end TEXT,
    general_waste_kg REAL,
    recycling_kg REAL,
    organic_waste_kg REAL,
    created_at TEXT NOT NULL,

    FOREIGN KEY (household_id)
        REFERENCES households(household_id)
);

------------------------------------------------------------
TRANSPORT ENTRIES
------------------------------------------------------------

CREATE TABLE transport_entries (
    transport_entry_id INTEGER PRIMARY KEY,
    household_id INTEGER NOT NULL,
    period_start TEXT NOT NULL,
    period_end TEXT,
    car_km REAL,
    public_transport_km REAL,
    cycling_km REAL,
    walking_km REAL,
    created_at TEXT NOT NULL,

    FOREIGN KEY (household_id)
        REFERENCES households(household_id)
);

------------------------------------------------------------
SCENARIOS
------------------------------------------------------------

CREATE TABLE scenarios (
    scenario_id INTEGER PRIMARY KEY,
    household_id INTEGER NOT NULL,
    name TEXT NOT NULL,
    description TEXT,
    created_by_user_id INTEGER,
    created_at TEXT NOT NULL,
    updated_at TEXT,

    FOREIGN KEY (household_id)
        REFERENCES households(household_id),

    FOREIGN KEY (created_by_user_id)
        REFERENCES users(user_id)
);
